# Optional sideload-only shared guest RAM. This is distinct from the JIT helper.
target_sources(CemuIOS PRIVATE SharedRAMHost.mm SharedRAMContext.mm SharedRAMMapping.mm)
add_library(CemuSharedRAMHelper MODULE SharedRAMHelper.mm SharedRAMContext.mm SharedRAMMapping.mm)
get_target_property(SHARED_RAM_TEAM CemuIOS XCODE_ATTRIBUTE_DEVELOPMENT_TEAM)
get_target_property(SHARED_RAM_SIGN_STYLE CemuIOS XCODE_ATTRIBUTE_CODE_SIGN_STYLE)
get_target_property(SHARED_RAM_SIGN_IDENTITY CemuIOS XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY)
set_target_properties(CemuSharedRAMHelper PROPERTIES
    BUNDLE TRUE BUNDLE_EXTENSION appex PREFIX "" SUFFIX ""
    XCODE_PRODUCT_TYPE "com.apple.product-type.app-extension"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/HelperInfo.plist"
    XCODE_ATTRIBUTE_INFOPLIST_FILE "${CMAKE_CURRENT_SOURCE_DIR}/HelperInfo.plist"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${CEMUIOS_BUNDLE_IDENTIFIER}.SharedRAMHelper"
    XCODE_ATTRIBUTE_PRODUCT_NAME "CemuSharedRAMHelper"
    XCODE_ATTRIBUTE_MARKETING_VERSION "${CEMUIOS_VERSION}"
    XCODE_ATTRIBUTE_CURRENT_PROJECT_VERSION "${CEMUIOS_BUILD_NUMBER}"
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "${CMAKE_OSX_DEPLOYMENT_TARGET}"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
    XCODE_ATTRIBUTE_CODE_SIGN_STYLE "${SHARED_RAM_SIGN_STYLE}"
    XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${SHARED_RAM_TEAM}"
    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "${SHARED_RAM_SIGN_IDENTITY}"
    XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES
    XCODE_ATTRIBUTE_APPLICATION_EXTENSION_API_ONLY YES
    XCODE_ATTRIBUTE_SKIP_INSTALL YES
    XCODE_ATTRIBUTE_OTHER_LDFLAGS "-e _NSExtensionMain"
    XCODE_ATTRIBUTE_LD_RUNPATH_SEARCH_PATHS "@executable_path/Frameworks @executable_path/../../Frameworks"
)
target_link_libraries(CemuSharedRAMHelper PRIVATE "-framework Foundation")
add_dependencies(CemuIOS CemuSharedRAMHelper)
set_property(TARGET CemuIOS APPEND PROPERTY XCODE_EMBED_APP_EXTENSIONS CemuSharedRAMHelper)
